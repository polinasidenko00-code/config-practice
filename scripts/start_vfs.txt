# Стартовый скрипт этапа 3: все реализованные команды,
# работа с VFS и обработка ошибок.
# Запускается из корня проекта: копии VFS сохраняются в out/.

# Команды-заглушки ls и cd
ls
ls -l /home /etc
cd

# Ошибки команд
cd /home /tmp

vfs-save out/vfs_copy

# непустая директория, файл вместо директории
vfs-save
vfs-save out/a out/b
vfs-save out/vfs_copy
vfs-save vfs/not_a_directory.txt
